/** * Voda Time Tracker — Apps Script backend. * * This stores all Time Tracker data: day entries, the vehicle list, the job * name list, and the employee list (used for the simple name-picker sign-in * — no Google account required). * * SETUP: * 1. Create a new Google Sheet. * 2. Extensions > Apps Script, paste this whole file in as Code.gs. * 3. Run `setupSheets` once from the Apps Script editor (select it in the * function dropdown, click Run) to create the sheet tabs this needs. * Safe to re-run later if you add new sheets to this file — it only * creates tabs that don't already exist. * 4. Deploy > New deployment > Web app. Execute as "Me", access "Anyone". * 5. Copy the deployment URL into CONFIG.API_URL in index.html. * * If you already deployed this once and are just adding the Employees * sheet: paste the updated code in, re-run `setupSheets`, then go to * Deploy > Manage deployments > pencil icon > New version > Deploy. That * keeps the same URL, so you don't need to change CONFIG.API_URL again. */ const SHEET_ENTRIES = 'Entries'; const SHEET_VEHICLES = 'Vehicles'; const SHEET_JOBNAMES = 'JobNames'; const SHEET_EMPLOYEES = 'Employees'; const SHEET_SETTINGS = 'Settings'; const SHEET_AUDITLOG = 'AuditLog'; const SHEET_SIGNOFFS = 'Signoffs'; function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss.getSheetByName(SHEET_ENTRIES)) { const sh = ss.insertSheet(SHEET_ENTRIES); sh.getRange(1, 1, 1, 2).setValues([['id', 'json']]); } if (!ss.getSheetByName(SHEET_VEHICLES)) { const sh = ss.insertSheet(SHEET_VEHICLES); sh.getRange(1, 1).setValue('[]'); } if (!ss.getSheetByName(SHEET_JOBNAMES)) { const sh = ss.insertSheet(SHEET_JOBNAMES); sh.getRange(1, 1).setValue('[]'); } if (!ss.getSheetByName(SHEET_EMPLOYEES)) { const sh = ss.insertSheet(SHEET_EMPLOYEES); sh.getRange(1, 1).setValue('[]'); } if (!ss.getSheetByName(SHEET_SETTINGS)) { const sh = ss.insertSheet(SHEET_SETTINGS); sh.getRange(1, 1).setValue('{"loginPin":"1234"}'); } if (!ss.getSheetByName(SHEET_AUDITLOG)) { const sh = ss.insertSheet(SHEET_AUDITLOG); sh.getRange(1, 1).setValue('[]'); } if (!ss.getSheetByName(SHEET_SIGNOFFS)) { const sh = ss.insertSheet(SHEET_SIGNOFFS); sh.getRange(1, 1).setValue('[]'); } } function doGet(e) { const action = e.parameter.action; if (action === 'getAll') { return respond({ entries: readEntries(), vehicles: readArraySheet(SHEET_VEHICLES), jobNames: readArraySheet(SHEET_JOBNAMES), employees: readArraySheet(SHEET_EMPLOYEES), settings: readObjectSheet(SHEET_SETTINGS), auditLog: readArraySheet(SHEET_AUDITLOG), signoffs: readArraySheet(SHEET_SIGNOFFS) }); } return respond({ error: 'Unknown action' }); } function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return respond({ error: 'Bad request body' }); } const action = body.action; const data = body.data; if (action === 'saveEntries') { writeEntries(data); return respond({ ok: true }); } if (action === 'saveVehicles') { writeArraySheet(SHEET_VEHICLES, data); return respond({ ok: true }); } if (action === 'saveJobNames') { writeArraySheet(SHEET_JOBNAMES, data); return respond({ ok: true }); } if (action === 'saveEmployees') { writeArraySheet(SHEET_EMPLOYEES, data); return respond({ ok: true }); } if (action === 'saveSettings') { writeObjectSheet(SHEET_SETTINGS, data); return respond({ ok: true }); } if (action === 'saveAuditLog') { writeArraySheet(SHEET_AUDITLOG, data); return respond({ ok: true }); } if (action === 'saveSignoffs') { writeArraySheet(SHEET_SIGNOFFS, data); return respond({ ok: true }); } return respond({ error: 'Unknown action' }); } /* ---- Entries: one row per day-entry, full JSON blob in column B ---- */ function readEntries() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_ENTRIES); const rows = sh.getDataRange().getValues(); const out = []; for (let i = 1; i < rows.length; i++) { const json = rows[i][1]; if (!json) continue; try { out.push(JSON.parse(json)); } catch (err) { /* skip corrupt row */ } } return out; } function writeEntries(entries) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_ENTRIES); sh.clearContents(); sh.getRange(1, 1, 1, 2).setValues([['id', 'json']]); if (!entries || !entries.length) return; const rows = entries.map(function (en) { return [en.id, JSON.stringify(en)]; }); sh.getRange(2, 1, rows.length, 2).setValues(rows); } /* ---- Vehicles / JobNames / Employees: whole array stored as JSON in cell A1 ---- */ function readArraySheet(name) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); const raw = sh.getRange(1, 1).getValue(); if (!raw) return []; try { return JSON.parse(raw); } catch (err) { return []; } } function writeArraySheet(name, arr) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); sh.getRange(1, 1).setValue(JSON.stringify(arr || [])); } /* ---- Settings: a single JSON object stored in cell A1 ---- */ function readObjectSheet(name) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); const raw = sh.getRange(1, 1).getValue(); if (!raw) return {}; try { return JSON.parse(raw); } catch (err) { return {}; } } function writeObjectSheet(name, obj) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); sh.getRange(1, 1).setValue(JSON.stringify(obj || {})); } function respond(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
